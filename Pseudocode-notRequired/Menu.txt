BEGIN PUBLIC CLASS Menu

    DECLARE PRIVATE BufferedReader consoleInputReader
    DECLARE PRIVATE SensorArray loadedReadings
    DECLARE PRIVATE STRING csvFilename


    CONSTRUCTOR Menu(SensorArray loadedReadings, STRING csvFilename)
        SET this.loadedReadings = loadedReadings
        SET this.csvFilename = csvFilename
        SET this.consoleInputReader = NEW BufferedReader(NEW InputStreamReader(System.in))
    END CONSTRUCTOR


    PUBLIC METHOD run()
        DECLARE BOOLEAN menuIsRunning = TRUE

        WHILE menuIsRunning EQUALS TRUE
            CALL showMainMenu()
            DECLARE INTEGER userChoice = CALL readIntegerInRange("choice: ", 1, 6)

            IF userChoice EQUALS 1 THEN
                CALL handleOverallStats()
            ELSE IF userChoice EQUALS 2 THEN
                CALL handleStatsByZone()
            ELSE IF userChoice EQUALS 3 THEN
                CALL handleStatsByType()
            ELSE IF userChoice EQUALS 4 THEN
                CALL handleAddReading()
            ELSE IF userChoice EQUALS 5 THEN
                CALL handleDeleteReading()
            ELSE IF userChoice EQUALS 6 THEN
                SET menuIsRunning = FALSE
            END IF
        END WHILE
    END METHOD


    PRIVATE METHOD showMainMenu()
        CALL Logger.logBlank()
        CALL Logger.log("Main menu")
        CALL Logger.log("1. overall statistics")
        CALL Logger.log("2. stats by zone")
        CALL Logger.log("3. stats by sensor type")
        CALL Logger.log("4. add a reading")
        CALL Logger.log("5. delete a reading")
        CALL Logger.log("6. exit")
    END METHOD


    PRIVATE METHOD handleOverallStats()
        CALL showStatsSubMenu(loadedReadings, "entire greenhouse")
    END METHOD


    PRIVATE METHOD handleStatsByZone()
        DECLARE STRING selectedZone = CALL readValidZone()
        DECLARE SensorArray filteredReadings = CALL loadedReadings.filterByZone(selectedZone)
        CALL showStatsSubMenu(filteredReadings, "zone " + selectedZone)
    END METHOD


    PRIVATE METHOD readValidZone() RETURNS STRING
        DECLARE STRING ARRAY zones = {"ZoneA", "ZoneB", "ZoneC"}

        CALL Logger.logBlank()
        CALL Logger.log("Select zone")

        FOR i = 0 TO zones.length - 1
            CALL Logger.log((i + 1) + ". " + zones[i])
        END FOR

        DECLARE INTEGER userSelection = CALL readIntegerInRange("zone: ", 1, zones.length)
        RETURN zones[userSelection - 1]
    END METHOD


    PRIVATE METHOD handleStatsByType()
        DECLARE STRING selectedSensorType = CALL readValidSensorType()
        DECLARE SensorArray filteredReadings = CALL loadedReadings.filterBySensorType(selectedSensorType)
        CALL showStatsSubMenu(filteredReadings, "type " + selectedSensorType)
    END METHOD


    PRIVATE METHOD readValidSensorType() RETURNS STRING
        DECLARE STRING ARRAY sensorTypes = {"temperature", "humidity", "soilMoisture", "light"}

        CALL Logger.logBlank()
        CALL Logger.log("Select sensor type")

        FOR i = 0 TO sensorTypes.length - 1
            CALL Logger.log((i + 1) + ". " + sensorTypes[i])
        END FOR

        DECLARE INTEGER userSelection = CALL readIntegerInRange("type: ", 1, sensorTypes.length)
        RETURN sensorTypes[userSelection - 1]
    END METHOD


    PRIVATE METHOD handleAddReading()
        CALL Logger.logBlank()
        CALL Logger.log("Add new reading")

        DECLARE STRING sensorID = CALL readNonEmptyString("sensor ID: ")

        WHILE loadedReadings.hasSensorID(sensorID) EQUALS TRUE
            CALL Logger.log("That sensor ID is already in the data.")
            SET sensorID = CALL readNonEmptyString("sensor ID: ")
        END WHILE

        DECLARE STRING sensorType = CALL readValidSensorType()
        DECLARE STRING zone = CALL readValidZone()
        DECLARE DOUBLE sensorValue = CALL readDoubleNumber("value: ")

        DECLARE INTEGER day = CALL readIntegerInRange("day (1-31): ", 1, 31)
        DECLARE INTEGER month = CALL readIntegerInRange("month (1-12): ", 1, 12)
        DECLARE INTEGER year = CALL readIntegerInRange("year (2000-2100): ", 2000, 2100)
        DECLARE INTEGER hour = CALL readIntegerInRange("hour (0-23): ", 0, 23)
        DECLARE INTEGER minute = CALL readIntegerInRange("minute (0-59): ", 0, 59)

        DECLARE Timestamp newReadingTimestamp = NEW Timestamp(day, month, year, hour, minute)
        DECLARE SensorReading newReading = NEW SensorReading(sensorID, sensorType, zone, sensorValue, newReadingTimestamp)

        CALL loadedReadings.add(newReading)
        CALL Logger.log("Added: " + newReading)
        CALL GreenhouseDataStorage.saveCSV(csvFilename, loadedReadings)
    END METHOD


    PRIVATE METHOD handleDeleteReading()
        IF loadedReadings.getCount() EQUALS 0 THEN
            CALL Logger.log("No readings to delete.")
        ELSE
            CALL Logger.logBlank()
            CALL Logger.log("Current readings")

            FOR i = 0 TO loadedReadings.getCount() - 1
                CALL Logger.log((i + 1) + ". " + loadedReadings.get(i))
            END FOR

            DECLARE INTEGER userSelection = CALL readIntegerInRange("delete number: ", 1, loadedReadings.getCount())
            DECLARE SensorReading removedReading = CALL loadedReadings.get(userSelection - 1)

            CALL loadedReadings.deleteAt(userSelection - 1)
            CALL Logger.log("Deleted: " + removedReading)
            CALL GreenhouseDataStorage.saveCSV(csvFilename, loadedReadings)
        END IF
    END METHOD


    PRIVATE METHOD showStatsSubMenu(SensorArray filteredReadings, STRING contextLabel)
        CALL Logger.logBlank()
        CALL Logger.log("Stats for " + contextLabel)
        CALL Logger.log("1. total readings")
        CALL Logger.log("2. average value")
        CALL Logger.log("3. minimum value")
        CALL Logger.log("4. maximum value")
        CALL Logger.log("5. number outside safe range")
        CALL Logger.log("6. percent outside safe range")
        CALL Logger.log("7. all of the above")

        DECLARE INTEGER userChoice = CALL readIntegerInRange("stat: ", 1, 7)
        CALL Logger.logBlank()

        IF userChoice EQUALS 1 THEN
            CALL Logger.log("Total: " + Math.totalReadings(filteredReadings))
        ELSE IF userChoice EQUALS 2 THEN
            CALL Logger.log("Average: " + Math.average(filteredReadings))
        ELSE IF userChoice EQUALS 3 THEN
            CALL Logger.log("Minimum: " + Math.minimum(filteredReadings))
        ELSE IF userChoice EQUALS 4 THEN
            CALL Logger.log("Maximum: " + Math.maximum(filteredReadings))
        ELSE IF userChoice EQUALS 5 THEN
            CALL Logger.log("Number outside safe range: " + Math.outOfRangeCount(filteredReadings))
        ELSE IF userChoice EQUALS 6 THEN
            CALL Logger.log("Percent outside safe range: " + Math.outOfRangePercent(filteredReadings))
        ELSE IF userChoice EQUALS 7 THEN
            CALL Logger.log("Total: " + Math.totalReadings(filteredReadings))
            CALL Logger.log("Average: " + Math.average(filteredReadings))
            CALL Logger.log("Minimum: " + Math.minimum(filteredReadings))
            CALL Logger.log("Maximum: " + Math.maximum(filteredReadings))
            CALL Logger.log("Number outside safe range: " + Math.outOfRangeCount(filteredReadings))
            CALL Logger.log("Percent outside safe range: " + Math.outOfRangePercent(filteredReadings))
        END IF
    END METHOD


    PRIVATE METHOD readLineFromConsole(STRING promptToShow) RETURNS STRING
        PRINT promptToShow WITHOUT NEW LINE

        TRY
            DECLARE STRING rawInputLine = READ LINE FROM consoleInputReader
            IF rawInputLine EQUALS NULL THEN
                RETURN ""
            ELSE
                RETURN rawInputLine.trim()
            END IF
        CATCH IOException
            RETURN ""
        END TRY
    END METHOD


    PRIVATE METHOD readIntegerInRange(STRING promptToShow, INTEGER minimumAllowedValue, INTEGER maximumAllowedValue) RETURNS INTEGER
        DECLARE INTEGER validatedInteger = 0
        DECLARE BOOLEAN inputAccepted = FALSE

        WHILE inputAccepted EQUALS FALSE
            DECLARE STRING userInputString = CALL readLineFromConsole(promptToShow)

            IF Validator.isInteger(userInputString) EQUALS TRUE THEN
                DECLARE INTEGER parsedInteger = PARSE userInputString AS INTEGER

                IF parsedInteger >= minimumAllowedValue AND parsedInteger <= maximumAllowedValue THEN
                    SET validatedInteger = parsedInteger
                    SET inputAccepted = TRUE
                ELSE
                    CALL Logger.log("Enter a number between " + minimumAllowedValue + " and " + maximumAllowedValue + ".")
                END IF
            ELSE
                CALL Logger.log("Enter a number between " + minimumAllowedValue + " and " + maximumAllowedValue + ".")
            END IF
        END WHILE

        RETURN validatedInteger
    END METHOD


    PRIVATE METHOD readDoubleNumber(STRING promptToShow) RETURNS DOUBLE
        DECLARE DOUBLE validatedDouble = 0.0
        DECLARE BOOLEAN inputAccepted = FALSE

        WHILE inputAccepted EQUALS FALSE
            DECLARE STRING userInputString = CALL readLineFromConsole(promptToShow)

            IF Validator.isNumeric(userInputString) EQUALS TRUE THEN
                SET validatedDouble = PARSE userInputString AS DOUBLE
                SET inputAccepted = TRUE
            ELSE
                CALL Logger.log("Enter a number.")
            END IF
        END WHILE

        RETURN validatedDouble
    END METHOD


    PRIVATE METHOD readNonEmptyString(STRING promptToShow) RETURNS STRING
        DECLARE STRING validatedString = ""
        DECLARE BOOLEAN inputAccepted = FALSE

        WHILE inputAccepted EQUALS FALSE
            DECLARE STRING userInputString = CALL readLineFromConsole(promptToShow)

            IF userInputString.length() > 0 THEN
                SET validatedString = userInputString
                SET inputAccepted = TRUE
            ELSE
                CALL Logger.log("This cannot be empty.")
            END IF
        END WHILE

        RETURN validatedString
    END METHOD

END CLASS
