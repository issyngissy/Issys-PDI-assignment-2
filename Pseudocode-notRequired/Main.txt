BEGIN PUBLIC CLASS Main

    DECLARE PRIVATE STATIC FINAL STRING CSV_FILE = "data.csv"

    PUBLIC STATIC METHOD main(args)
        CALL Logger.open()

        DECLARE SensorArray loadedReadings
        SET loadedReadings = CALL GreenhouseDataStorage.loadCSV(CSV_FILE)

        DECLARE Menu menu
        SET menu = NEW Menu(loadedReadings, CSV_FILE)

        CALL menu.run()

        CALL Logger.close()
    END METHOD

END CLASS
