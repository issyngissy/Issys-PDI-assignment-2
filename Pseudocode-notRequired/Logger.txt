BEGIN PUBLIC CLASS Logger

    DECLARE PRIVATE STATIC FINAL STRING LOG_FILE_NAME = "logged_data"
    DECLARE PRIVATE STATIC BufferedWriter logFileWriter = NULL


    PUBLIC STATIC METHOD open()
        TRY
            SET logFileWriter = OPEN LOG_FILE_NAME FOR APPENDING
            WRITE "starting session..." TO logFileWriter
            WRITE NEW LINE TO logFileWriter
            FLUSH logFileWriter
        CATCH Exception
            PRINT "could not open log file..."
            SET logFileWriter = NULL
        END TRY
    END METHOD


    PUBLIC STATIC METHOD log(STRING messageToLog)
        PRINT messageToLog

        IF logFileWriter IS NOT NULL THEN
            TRY
                WRITE messageToLog TO logFileWriter
                WRITE NEW LINE TO logFileWriter
                FLUSH logFileWriter
            CATCH IOException
                DO NOTHING
            END TRY
        END IF
    END METHOD


    PUBLIC STATIC METHOD logBlank()
        CALL log("")
    END METHOD


    PUBLIC STATIC METHOD close()
        IF logFileWriter IS NOT NULL THEN
            TRY
                WRITE "ending session" TO logFileWriter
                WRITE NEW LINE TO logFileWriter
                WRITE NEW LINE TO logFileWriter
                FLUSH logFileWriter
                CLOSE logFileWriter
            CATCH IOException
                DO NOTHING
            END TRY
        END IF
    END METHOD

END CLASS
